function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. VALIDAR CÓDIGO Y MARCAR COMO USADO INMEDIATAMENTE AL ESCANEAR if (action === "validarCodigo") { const codigo = e.parameter.codigo; if (!codigo) { return respuestaJSON({ status: "ERROR", message: "Código no proporcionado" }); } const sheetCodigos = ss.getSheetByName("Codigos_Emitidos"); const dataCodigos = sheetCodigos.getDataRange().getValues(); for (let i = 1; i < dataCodigos.length; i++) { const codigoFila = dataCodigos[i][0].toString().trim().toUpperCase(); const estadoFila = dataCodigos[i][2].toString().trim().toUpperCase(); if (codigoFila === codigo.trim().toUpperCase()) { if (estadoFila === "DISPONIBLE") { // Marca automáticamente como USADO en la fila y registra la fecha/hora sheetCodigos.getRange(i + 1, 3).setValue("USADO"); sheetCodigos.getRange(i + 1, 4).setValue(new Date()); return respuestaJSON({ status: "OK", message: "Código válido y registrado" }); } else { return respuestaJSON({ status: "ERROR", message: "Este ticket ya fue utilizado" }); } } } return respuestaJSON({ status: "ERROR", message: "Código no encontrado" }); } // 2. REGISTRAR EL PUNTAJE FINAL DE LA PARTIDA EN PARTIDAS_REGISTRADAS if (action === "registrarPartida") { const codigo = e.parameter.codigo; const puntos = e.parameter.puntos; const sheetPartidas = ss.getSheetByName("Partidas_Registradas"); const idPartida = "PART-" + new Date().getTime(); sheetPartidas.appendRow([idPartida, codigo, puntos, new Date()]); return respuestaJSON({ status: "OK", idPartida: idPartida }); } return respuestaJSON({ status: "ERROR", message: "Acción no válida" }); } // Función auxiliar para retornar respuestas en formato JSON function respuestaJSON(objeto) { return ContentService.createTextOutput(JSON.stringify(objeto)) .setMimeType(ContentService.MimeType.JSON); } // GENERADOR MASIVO DE CÓDIGOS DE BARRAS function generarCodigosMasivos() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Codigos_Emitidos"); for (let i = 0; i < 50; i++) { const codigoRandom = "SEL-" + Math.random().toString(36).substring(2, 7).toUpperCase(); const formulaBarcode = '=IMAGE("https://barcode.tec-it.com/barcode.ashx?data=' + codigoRandom + '&code=Code128")'; sheet.appendRow([codigoRandom, formulaBarcode, "DISPONIBLE", ""]); } }
