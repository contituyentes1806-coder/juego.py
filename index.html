function doGet(e) { var params = e.parameter; var action = params.action; var codigo = params.codigo; if (!codigo) { return responderJSON({ status: "ERROR", message: "Falta el parámetro código" }); } var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); var codigoBuscado = codigo.toString().trim().toUpperCase(); // --- ACCIÓN 1: VALIDAR TICKET ANTES DE JUGAR --- if (action === "validarCodigo") { for (var i = 1; i < data.length; i++) { var codigoFila = data[i][0].toString().trim().toUpperCase(); if (codigoFila === codigoBuscado) { var estado = data[i][1] ? data[i][1].toString().trim().toUpperCase() : "DISPONIBLE"; if (estado === "USADO") { return responderJSON({ status: "ERROR", message: "EL TICKET YA FUE UTILIZADO" }); } return responderJSON({ status: "OK", message: "Código válido" }); } } return responderJSON({ status: "ERROR", message: "TICKET INEXISTENTE O INVÁLIDO" }); } // --- ACCIÓN 2: REGISTRAR PARTIDA Y MARCAR COMO USADO --- if (action === "registrarPartida") { var puntos = parseInt(params.puntos) || 0; var premioObtenido = "Siga participando"; // Lógica para asignar premios según puntaje if (puntos >= 150) { premioObtenido = "CORTESÍA CAFE EXTRA GNC"; } for (var j = 1; j < data.length; j++) { var cdg = data[j][0].toString().trim().toUpperCase(); if (cdg === codigoBuscado) { // Actualiza el estado a USADO y guarda el resultado sheet.getRange(j + 1, 2).setValue("USADO"); sheet.getRange(j + 1, 3).setValue(puntos); sheet.getRange(j + 1, 4).setValue(premioObtenido); sheet.getRange(j + 1, 5).setValue(new Date()); return responderJSON({ status: "OK", message: "Partida registrada", premioAsignado: premioObtenido }); } } return responderJSON({ status: "ERROR", message: "No se pudo registrar la partida" }); } return responderJSON({ status: "ERROR", message: "Acción no reconocida" }); } function responderJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
