    <script>
        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x111625);
        scene.fog = new THREE.FogExp2(0x111625, 0.015);

        const aspect = window.innerWidth / window.innerHeight;
        const camera = new THREE.PerspectiveCamera(aspect < 1 ? 75 : 60, aspect, 0.1, 1000);
        camera.position.set(0, 3, 7);
        camera.lookAt(0, 1, -10);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        renderer.shadowMap.enabled = true;
        container.appendChild(renderer.domElement);

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xffffff, 0.9);
        dirLight.position.set(10, 20, 10);
        dirLight.castShadow = true;
        scene.add(dirLight);

        const roadGeometry = new THREE.PlaneGeometry(12, 400);
        const roadMaterial = new THREE.MeshStandardMaterial({ color: 0x333338, roughness: 0.8 });
        const road = new THREE.Mesh(roadGeometry, roadMaterial);
        road.rotation.x = -Math.PI / 2;
        road.position.z = -180;
        road.receiveShadow = true;
        scene.add(road);

        const lineGeo = new THREE.PlaneGeometry(0.3, 400);
        const lineMat = new THREE.MeshBasicMaterial({ color: 0xffcc00 });
        const leftLine = new THREE.Mesh(lineGeo, lineMat);
        leftLine.rotation.x = -Math.PI / 2;
        leftLine.position.set(-3.5, 0.01, -180);
        scene.add(leftLine);

        const rightLine = leftLine.clone();
        rightLine.position.x = 3.5;
        scene.add(rightLine);

        let juegoIniciado = false;
        let nivel = 1;
        let puntos = 0;
        let metaPuntosNivel = 100;
        let vidas = 3;

        let velMinKm = 80;
        let velMaxKm = 220;
        let tasaAceleracion = 1.5;
        let velActualKm = 0;

        let targetX = 0;
        let enTrompo = false;
        let anguloTrompoAcumulado = 0;

        const keys = { left: false, right: false, accel: false, brake: false };

        const playerGroup = new THREE.Group();
        scene.add(playerGroup);

        function buildFallbackPlayerCar() {
            const geo = new THREE.BoxGeometry(1.2, 0.6, 2.5);
            const mat = new THREE.MeshStandardMaterial({ color: 0x00ffcc });
            const mesh = new THREE.Mesh(geo, mat);
            mesh.position.y = 0.3;
            playerGroup.add(mesh);
        }

        const gltfLoader = new THREE.GLTFLoader();
        gltfLoader.load('/static/player_car.glb', (gltf) => {
            const model = gltf.scene;
            model.scale.set(0.25, 0.25, 0.25);
            const box = new THREE.Box3().setFromObject(model);
            const center = box.getCenter(new THREE.Vector3());
            model.position.sub(center);
            model.rotation.y = Math.PI; // Orienta la parte trasera hacia la cámara
            model.position.y = 0.1;
            model.traverse((node) => { if (node.isMesh) node.castShadow = true; });
            playerGroup.add(model);
        }, undefined, () => { buildFallbackPlayerCar(); });

        const items = [];
        const obstaculos = [];
        const gncGeo = new THREE.CylinderGeometry(0.35, 0.35, 0.8, 16);
        const gncMat = new THREE.MeshStandardMaterial({ color: 0x00ccff, metalness: 0.6 });
        const bonusGeo = new THREE.CylinderGeometry(0.4, 0.4, 0.1, 16);
        const bonusMat = new THREE.MeshStandardMaterial({ color: 0xffd700, metalness: 0.8 });
        const obsGeo = new THREE.ConeGeometry(0.45, 0.9, 16);
        const obsMat = new THREE.MeshStandardMaterial({ color: 0xff2200, roughness: 0.3 });
        const carriles = [-2, 0, 2];

        function generarObjeto() {
            if (!juegoIniciado || enTrompo) return;

            const rand = Math.random();
            const xPos = carriles[Math.floor(Math.random() * carriles.length)];

            if (rand < 0.35 + (nivel * 0.05)) {
                const obs = new THREE.Mesh(obsGeo, obsMat);
                obs.position.set(xPos, 0.45, -150);
                obs.castShadow = true;
                scene.add(obs);
                obstaculos.push(obs);
            } else if (rand < 0.85) {
                const item = new THREE.Mesh(gncGeo, gncMat);
                item.position.set(xPos, 0.4, -150);
                item.castShadow = true;
                item.userData = { tipo: 'gnc', puntos: 15 };
                scene.add(item);
                items.push(item);
            } else {
                const bonus = new THREE.Mesh(bonusGeo, bonusMat);
                bonus.rotation.x = Math.PI / 2;
                bonus.position.set(xPos, 0.5, -150);
                bonus.castShadow = true;
                bonus.userData = { tipo: 'bonus', puntos: 30 };
                scene.add(bonus);
                items.push(bonus);
            }
        }

        setInterval(generarObjeto, 900);

        function setKeyState(e, state) {
            const k = e.key;
            const c = e.code;

            if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(k) ||
                ['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space'].includes(c)) {
                e.preventDefault();
            }

            if (k === 'ArrowLeft' || c === 'ArrowLeft' || k === 'a' || k === 'A' || c === 'KeyA') keys.left = state;
            if (k === 'ArrowRight' || c === 'ArrowRight' || k === 'd' || k === 'D' || c === 'KeyD') keys.right = state;
            if (k === 'ArrowUp' || c === 'ArrowUp' || k === 'w' || k === 'W' || c === 'KeyW') keys.accel = state;
            if (k === 'ArrowDown' || c === 'ArrowDown' || k === 's' || k === 'S' || c === 'KeyS') keys.brake = state;
        }

        window.addEventListener('keydown', (e) => setKeyState(e, true), false);
        window.addEventListener('keyup', (e) => setKeyState(e, false), false);

        const bindTouch = (id, key) => {
            const el = document.getElementById(id);
            if (!el) return;
            const startHandler = (e) => { e.preventDefault(); keys[key] = true; };
            const endHandler = (e) => { e.preventDefault(); keys[key] = false; };
            el.addEventListener('touchstart', startHandler, { passive: false });
            el.addEventListener('touchend', endHandler, { passive: false });
            el.addEventListener('mousedown', () => { keys[key] = true; });
            el.addEventListener('mouseup', () => { keys[key] = false; });
        };

        bindTouch('btn-left', 'left');
        bindTouch('btn-right', 'right');
        bindTouch('btn-accel', 'accel');
        bindTouch('btn-brake', 'brake');

        function prepararJuego() {
            const parseNum = (id, fallback) => {
                const rawVal = document.getElementById(id).value;
                const cleanVal = String(rawVal).replace(',', '.');
                const parsed = parseFloat(cleanVal);
                return isNaN(parsed) ? fallback : parsed;
            };

            velMinKm = parseNum('cfg-speed-init', 80);
            velMaxKm = parseNum('cfg-speed-max', 220);
            tasaAceleracion = parseNum('cfg-accel', 1.5);
            velActualKm = velMinKm;

            document.getElementById('config-overlay').style.display = 'none';
            document.getElementById('game-over-overlay').style.display = 'none';
            document.getElementById('touch-controls').style.display = 'flex';

            juegoIniciado = true;
            window.focus();
            actualizarHUD();
            animate();
        }

        function reiniciarJuego() {
            items.forEach(item => scene.remove(item));
            obstaculos.forEach(obs => scene.remove(obs));
            items.length = 0;
            obstaculos.length = 0;

            nivel = 1;
            puntos = 0;
            metaPuntosNivel = 100;
            vidas = 3;
            targetX = 0;
            playerGroup.position.x = 0;
            playerGroup.rotation.y = 0;
            enTrompo = false;
            anguloTrompoAcumulado = 0;

            document.getElementById('config-overlay').style.display = 'flex';
            document.getElementById('game-over-overlay').style.display = 'none';
        }

        function actualizarHUD() {
            document.getElementById('nivel-display').innerText = `NIVEL: ${nivel}`;
            document.getElementById('puntos-display').innerText = `PTS: ${puntos} / ${metaPuntosNivel}`;
            document.getElementById('vidas-display').innerText = '❤️ '.repeat(Math.max(0, vidas));
            document.getElementById('speed-val').innerText = Math.round(velActualKm);
        }

        function activarEfectoTrompo() {
            enTrompo = true;
            anguloTrompoAcumulado = 0;
            vidas--;
            velActualKm = Math.max(velMinKm * 0.4, velActualKm * 0.3);
            actualizarHUD();
            flashDano();

            if (vidas <= 0) {
                setTimeout(() => finalizarJuego(), 600);
            }
        }

        function flashDano() {
            const hud = document.getElementById('hud');
            hud.classList.add('hud-damage');
            setTimeout(() => hud.classList.remove('hud-damage'), 400);
        }

        function actualizarFisicas() {
            if (!juegoIniciado) return;

            if (keys.accel) {
                velActualKm = Math.min(velMaxKm, velActualKm + tasaAceleracion);
            } else if (keys.brake) {
                velActualKm = Math.max(velMinKm * 0.3, velActualKm - (tasaAceleracion * 2.5));
            } else {
                if (velActualKm > velMinKm) velActualKm -= 0.5;
                if (velActualKm < velMinKm) velActualKm += 0.5;
            }

            document.getElementById('speed-val').innerText = Math.round(velActualKm);

            if (!enTrompo) {
                if (keys.left) targetX = Math.max(-2.5, targetX - 0.12);
                if (keys.right) targetX = Math.min(2.5, targetX + 0.12);
                playerGroup.position.x += (targetX - playerGroup.position.x) * 0.15;
            } else {
                const pasoRotacion = 0.25;
                anguloTrompoAcumulado += pasoRotacion;
                playerGroup.rotation.y = Math.PI + anguloTrompoAcumulado;

                if (anguloTrompoAcumulado >= Math.PI * 2) {
                    playerGroup.rotation.y = 0;
                    anguloTrompoAcumulado = 0;
                    enTrompo = false;
                }
            }

            const factorVelocidad = (velActualKm / 100) * 0.6;

            for (let i = items.length - 1; i >= 0; i--) {
                const item = items[i];
                item.position.z += factorVelocidad;
                item.rotation.y += 0.05;

                if (playerGroup.position.distanceTo(item.position) < 1.3 && !enTrompo) {
                    puntos += item.userData.puntos;
                    scene.remove(item);
                    items.splice(i, 1);

                    if (puntos >= metaPuntosNivel) {
                        nivel++;
                        metaPuntosNivel += 120;
                        velMaxKm += 20;
                    }
                    actualizarHUD();
                } else if (item.position.z > 10) {
                    scene.remove(item);
                    items.splice(i, 1);
                }
            }

            for (let j = obstaculos.length - 1; j >= 0; j--) {
                const obs = obstaculos[j];
                obs.position.z += factorVelocidad;

                if (playerGroup.position.distanceTo(obs.position) < 1.1 && !enTrompo) {
                    scene.remove(obs);
                    obstaculos.splice(j, 1);
                    activarEfectoTrompo();
                } else if (obs.position.z > 10) {
                    scene.remove(obs);
                    obstaculos.splice(j, 1);
                }
            }
        }

        let animFrameId = null;
        function animate() {
            if (!juegoIniciado) return;
            animFrameId = requestAnimationFrame(animate);
            actualizarFisicas();
            renderer.render(scene, camera);
        }

        function finalizarJuego() {
            juegoIniciado = false;
            if (animFrameId) cancelAnimationFrame(animFrameId);
            document.getElementById('touch-controls').style.display = 'none';
            const overlay = document.getElementById('game-over-overlay');
            overlay.style.display = 'flex';
            
            document.getElementById('over-title').innerText = "¡SIN VIDAS!";
            document.getElementById('over-msg').innerText = `Puntaje final: ${puntos} puntos (Nivel ${nivel}).`;
        }

        window.addEventListener('resize', () => {
            const newAspect = window.innerWidth / window.innerHeight;
            camera.aspect = newAspect;
            camera.fov = newAspect < 1 ? 75 : 60;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });
    </script>
